import json
import subprocess
import sys
import time
from pathlib import Path
import requests

base = 'http://gateway:8000'
selected = requests.get(base+'/api/v1/catalogue',timeout=90).json()['selections'][0]
records=[]
for iteration in range(24):
    submitted=subprocess.run([sys.executable,'-m','publication.worker','submit','--clock','2026-10-19T05:00:00Z',
                              '--hours',str(iteration+1)],capture_output=True)
    assert submitted.returncode==0
    worker=subprocess.Popen([sys.executable,'-m','publication.worker','once'],stdout=subprocess.PIPE,stderr=subprocess.PIPE)
    try:
        while worker.poll() is None:
            point_query={key:selected[key] for key in ('selection','run','revision','valid_time')}
            point_query.update(layer=selected['field'],lon=-23.5,lat=29.5)
            for path,params in [('/api/v1/catalogue',{}),(selected['point'],point_query),(selected['tiles'],{}),
                                ('/api/v1/selections/'+selected['selection']+'/tiles/2/3/4/2.f32',{})]:
                response=requests.get(base+path,params=params,timeout=90)
                records.append({'iteration':iteration,'path':path,'status':response.status_code,'request_id':response.headers.get('X-Request-ID')})
                if response.status_code != 200:
                    records[-1]['body']=response.json()
                    raise AssertionError(records[-1])
        stdout,stderr=worker.communicate()
        assert worker.returncode==0,(worker.returncode,stderr.decode()[:1000])
    finally:
        if worker.poll() is None:
            worker.kill()
            worker.communicate()
        Path('/evidence/migration-probe.json').write_text(json.dumps(records,indent=2))
print(json.dumps({'iterations':24,'requests':len(records),'status':'passed'}))
